/*
  Arduino Due DAC0 beatless sum-of-two-sines generator.

  Serial commands, 115200 baud, newline ending:
    PING
    CONFIG <f1_hz> <a1> <phase1_deg> <f2_hz> <a2> <phase2_deg> <out_vpp> <offset_v> <epsilon>
    START
    STOP
    STATUS
    HELP

  The corrected signal is a generalized version of theory/Sum_of_sin_corrected:

    raw(t) = a1*sin(theta1) + a2*sin(theta2)
    envelope(t) = sqrt(a1^2 + a2^2 + 2*a1*a2*cos(theta2 - theta1))
    normalized(t) = raw(t) / max(envelope(t), epsilon)
    vout(t) = offset_v + out_vpp/2 * normalized(t)

  For equal amplitudes this is equivalent to dividing the sum by the beating
  envelope |cos(pi*(f2-f1)*t + phase_delta/2)|, then scaling to DAC range.

  Arduino Due DAC0 is not rail-to-rail. Typical safe output range is about
  0.55..2.75 V.
*/

#include <Arduino.h>
#include <math.h>
#include <stdlib.h>
#include <string.h>

static const uint32_t BAUD_RATE = 115200;
static const uint16_t DAC_MAX_CODE = 4095;
static const float DAC_VMIN = 0.55f;
static const float DAC_VMAX = 2.75f;
static const float DEFAULT_OFFSET_V = 1.65f;
static const uint32_t UPDATE_US = 1000;

struct PortBuffer {
  char line[192];
  size_t len;
};

struct GeneratorConfig {
  float f1Hz;
  float a1;
  float phase1Deg;
  float f2Hz;
  float a2;
  float phase2Deg;
  float outputVpp;
  float offsetV;
  float epsilon;
};

static PortBuffer serialBuffer = {{0}, 0};
static PortBuffer serialUsbBuffer = {{0}, 0};
static GeneratorConfig config = {10.1f, 1.0f, 0.0f, 10.6f, 1.0f, 0.0f, 1.8f, DEFAULT_OFFSET_V, 0.02f};
static bool running = false;
static uint32_t startUs = 0;
static uint32_t lastUpdateUs = 0;
static uint16_t currentCode = 0;

static float clampFloat(float value, float low, float high) {
  if (value < low) {
    return low;
  }
  if (value > high) {
    return high;
  }
  return value;
}

static uint16_t voltageToCode(float voltage) {
  float clipped = clampFloat(voltage, DAC_VMIN, DAC_VMAX);
  float normalized = (clipped - DAC_VMIN) / (DAC_VMAX - DAC_VMIN);
  long code = static_cast<long>(normalized * DAC_MAX_CODE + 0.5f);
  if (code < 0) {
    return 0;
  }
  if (code > DAC_MAX_CODE) {
    return DAC_MAX_CODE;
  }
  return static_cast<uint16_t>(code);
}

static float codeToVoltage(uint16_t code) {
  return DAC_VMIN + (DAC_VMAX - DAC_VMIN) * (static_cast<float>(code) / static_cast<float>(DAC_MAX_CODE));
}

static void writeVoltage(float voltage) {
  currentCode = voltageToCode(voltage);
  analogWrite(DAC0, currentCode);
}

static bool configIsValid(const GeneratorConfig &candidate) {
  if (candidate.f1Hz < 0.0f || candidate.f1Hz > 100.0f || candidate.f2Hz < 0.0f || candidate.f2Hz > 100.0f) {
    return false;
  }
  if (candidate.a1 < 0.0f || candidate.a2 < 0.0f || candidate.a1 + candidate.a2 <= 0.0f) {
    return false;
  }
  if (candidate.outputVpp < 0.0f || candidate.outputVpp > (DAC_VMAX - DAC_VMIN)) {
    return false;
  }
  if (candidate.offsetV < DAC_VMIN || candidate.offsetV > DAC_VMAX) {
    return false;
  }
  if (candidate.offsetV - candidate.outputVpp * 0.5f < DAC_VMIN || candidate.offsetV + candidate.outputVpp * 0.5f > DAC_VMAX) {
    return false;
  }
  if (candidate.epsilon < 0.001f || candidate.epsilon > 1.0f) {
    return false;
  }
  return true;
}

static float correctedSample(float t) {
  float theta1 = TWO_PI * config.f1Hz * t + config.phase1Deg * DEG_TO_RAD;
  float theta2 = TWO_PI * config.f2Hz * t + config.phase2Deg * DEG_TO_RAD;
  float s1 = sinf(theta1);
  float s2 = sinf(theta2);
  float raw = config.a1 * s1 + config.a2 * s2;
  float envelopeSquared =
      config.a1 * config.a1 +
      config.a2 * config.a2 +
      2.0f * config.a1 * config.a2 * cosf(theta2 - theta1);
  float envelope = sqrtf(max(0.0f, envelopeSquared));
  float denominator = max(envelope, config.epsilon);
  float normalized = clampFloat(raw / denominator, -1.0f, 1.0f);
  return config.offsetV + config.outputVpp * 0.5f * normalized;
}

static void serviceGenerator() {
  if (!running) {
    return;
  }

  uint32_t now = micros();
  if (now - lastUpdateUs < UPDATE_US) {
    return;
  }
  lastUpdateUs = now;

  float t = static_cast<float>(now - startUs) / 1000000.0f;
  writeVoltage(correctedSample(t));
}

static void printConfigObject(Print &out) {
  out.print("\"f1_hz\":");
  out.print(config.f1Hz, 6);
  out.print(",\"a1\":");
  out.print(config.a1, 6);
  out.print(",\"phase1_deg\":");
  out.print(config.phase1Deg, 3);
  out.print(",\"f2_hz\":");
  out.print(config.f2Hz, 6);
  out.print(",\"a2\":");
  out.print(config.a2, 6);
  out.print(",\"phase2_deg\":");
  out.print(config.phase2Deg, 3);
  out.print(",\"output_vpp\":");
  out.print(config.outputVpp, 6);
  out.print(",\"offset_v\":");
  out.print(config.offsetV, 6);
  out.print(",\"epsilon\":");
  out.print(config.epsilon, 6);
}

static void printStatus(Print &out, const char *type) {
  out.print("{\"type\":\"");
  out.print(type);
  out.print("\",\"running\":");
  out.print(running ? "true" : "false");
  out.print(",\"dac\":\"DAC0\",\"dac_code\":");
  out.print(currentCode);
  out.print(",\"dac_voltage_v\":");
  out.print(codeToVoltage(currentCode), 6);
  out.print(",");
  printConfigObject(out);
  out.println("}");
}

static void printError(Print &out, const char *message) {
  out.print("{\"type\":\"error\",\"message\":\"");
  out.print(message);
  out.println("\"}");
}

static void printHelp(Print &out) {
  out.println("{\"type\":\"help\",\"commands\":[\"PING\",\"CONFIG <f1_hz> <a1> <phase1_deg> <f2_hz> <a2> <phase2_deg> <out_vpp> <offset_v> <epsilon>\",\"START\",\"STOP\",\"STATUS\",\"HELP\"]}");
}

static bool parseConfig(GeneratorConfig *candidate) {
  char *args[9] = {};
  for (int i = 0; i < 9; ++i) {
    args[i] = strtok(NULL, " \t\r\n");
    if (args[i] == NULL) {
      return false;
    }
  }

  candidate->f1Hz = atof(args[0]);
  candidate->a1 = atof(args[1]);
  candidate->phase1Deg = atof(args[2]);
  candidate->f2Hz = atof(args[3]);
  candidate->a2 = atof(args[4]);
  candidate->phase2Deg = atof(args[5]);
  candidate->outputVpp = atof(args[6]);
  candidate->offsetV = atof(args[7]);
  candidate->epsilon = atof(args[8]);
  return configIsValid(*candidate);
}

static void handleCommand(char *line, Print &out) {
  char *command = strtok(line, " \t\r\n");
  if (command == NULL) {
    return;
  }

  if (!strcasecmp(command, "PING")) {
    out.println("{\"type\":\"pong\",\"board\":\"Arduino Due\",\"mode\":\"sum_sine_beatless\",\"dac\":\"DAC0\",\"dac_vmin\":0.55,\"dac_vmax\":2.75}");
    return;
  }

  if (!strcasecmp(command, "HELP")) {
    printHelp(out);
    return;
  }

  if (!strcasecmp(command, "STATUS")) {
    printStatus(out, "status");
    return;
  }

  if (!strcasecmp(command, "START")) {
    running = true;
    startUs = micros();
    lastUpdateUs = 0;
    printStatus(out, "started");
    return;
  }

  if (!strcasecmp(command, "STOP")) {
    running = false;
    writeVoltage(config.offsetV);
    printStatus(out, "stopped");
    return;
  }

  if (!strcasecmp(command, "CONFIG")) {
    GeneratorConfig candidate = config;
    if (!parseConfig(&candidate)) {
      printError(out, "CONFIG expects valid f1 a1 phase1 f2 a2 phase2 out_vpp offset_v epsilon; output must stay in 0.55..2.75 V");
      return;
    }
    config = candidate;
    startUs = micros();
    lastUpdateUs = 0;
    writeVoltage(config.offsetV);
    printStatus(out, "configured");
    return;
  }

  printError(out, "unknown command");
}

static void servicePort(Stream &input, Print &output, PortBuffer &buffer) {
  while (input.available() > 0) {
    char ch = static_cast<char>(input.read());
    if (ch == '\n' || ch == '\r') {
      if (buffer.len > 0) {
        buffer.line[buffer.len] = '\0';
        handleCommand(buffer.line, output);
        buffer.len = 0;
      }
      continue;
    }

    if (buffer.len < sizeof(buffer.line) - 1) {
      buffer.line[buffer.len++] = ch;
    } else {
      buffer.len = 0;
      printError(output, "line too long");
    }
  }
}

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, LOW);

  analogWriteResolution(12);
  pinMode(DAC0, OUTPUT);
  writeVoltage(DEFAULT_OFFSET_V);

  Serial.begin(BAUD_RATE);
  SerialUSB.begin(BAUD_RATE);
}

void loop() {
  servicePort(Serial, Serial, serialBuffer);
  servicePort(SerialUSB, SerialUSB, serialUsbBuffer);
  serviceGenerator();

  static uint32_t lastBlinkMs = 0;
  uint32_t nowMs = millis();
  if (nowMs - lastBlinkMs >= (running ? 250 : 1000)) {
    lastBlinkMs = nowMs;
    digitalWrite(LED_BUILTIN, !digitalRead(LED_BUILTIN));
  }
}
