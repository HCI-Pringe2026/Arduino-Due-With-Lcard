/**
 * dual_sine_dac.ino
 * Arduino Due — DAC0 dual-sine wave generator
 *
 * Formula: V(t) = DC + A1*sin(2π*f1*t + φ1) + A2*sin(2π*f2*t + φ2)
 * Output is clamped to [0, 3.3V] then mapped to 12-bit DAC (0–4095).
 *
 * Serial protocol (115200 baud, newline-terminated):
 *   SET A1 <float>   — Amplitude 1 in Volts  (0.0 – 1.65)
 *   SET F1 <float>   — Frequency 1 in Hz     (0.0 – 5000.0)
 *   SET P1 <float>   — Phase 1 in degrees    (any float)
 *   SET A2 <float>   — Amplitude 2 in Volts
 *   SET F2 <float>   — Frequency 2 in Hz
 *   SET P2 <float>   — Phase 2 in degrees
 *   SET DC <float>   — DC offset in Volts    (0.0 – 3.3)
 *   SET SR <int>     — Sample rate in Hz     (100 – 100000)
 *   START            — Begin output
 *   STOP             — Halt output (DAC set to 0)
 *   STATUS           — Print current parameters as JSON
 *
 * Safeguards:
 *   • All values are clamped server-side before writing to DAC
 *   • A1+A2+|DC| is checked; amplitudes are auto-scaled if they would clip
 *   • Sample rate capped at 100 kHz (Due DAC limit)
 *   • Timer interrupt drives output — no busy-wait
 */

#include <Arduino.h>

// ── Constants ────────────────────────────────────────────────────────────────
static const float DAC_VREF      = 3.3f;
static const int   DAC_BITS      = 4095;        // 12-bit
static const float DAC_MID_V     = 1.65f;
static const float MAX_AMPLITUDE = 1.65f;       // per sine, worst-case headroom
static const float MAX_DC        = 3.3f;
static const float MAX_FREQ      = 5000.0f;     // Hz — above this waveform quality degrades
static const int   MIN_SR        = 100;
static const int   MAX_SR        = 100000;
static const int   CMD_BUF_SIZE  = 64;

// ── Parameters (volatile — shared with ISR) ──────────────────────────────────
volatile float  g_amp1   = 0.5f;      // V
volatile float  g_freq1  = 100.0f;    // Hz
volatile float  g_phase1 = 0.0f;      // radians

volatile float  g_amp2   = 0.5f;
volatile float  g_freq2  = 200.0f;
volatile float  g_phase2 = 0.0f;

volatile float  g_dc     = 1.65f;     // V

volatile int    g_sampleRate = 10000; // Hz
volatile bool   g_running    = false;

// ── ISR state ────────────────────────────────────────────────────────────────
volatile uint32_t g_sampleCount = 0;
volatile float    g_dt          = 1.0f / 10000.0f;

// ── Timer2 ISR — runs at g_sampleRate ────────────────────────────────────────
void TC6_Handler(void) {
    TC_GetStatus(TC2, 0);   // clear interrupt flag

    if (!g_running) {
        analogWrite(DAC0, 0);
        return;
    }

    float t = g_sampleCount * g_dt;
    g_sampleCount++;

    float v = g_dc
            + g_amp1 * sinf(TWO_PI * g_freq1 * t + g_phase1)
            + g_amp2 * sinf(TWO_PI * g_freq2 * t + g_phase2);

    // Clamp to safe DAC range
    if (v < 0.0f)      v = 0.0f;
    if (v > DAC_VREF)  v = DAC_VREF;

    uint16_t dac_val = (uint16_t)((v / DAC_VREF) * DAC_BITS + 0.5f);
    if (dac_val > DAC_BITS) dac_val = DAC_BITS;

    analogWrite(DAC0, dac_val);
}

// ── Timer setup ──────────────────────────────────────────────────────────────
void setupTimer(int sampleRate) {
    sampleRate = constrain(sampleRate, MIN_SR, MAX_SR);
    g_sampleRate = sampleRate;
    g_dt = 1.0f / (float)sampleRate;
    g_sampleCount = 0;

    // TC2, channel 0 → IRQ TC6
    pmc_set_writeprotect(false);
    pmc_enable_periph_clk(TC6_IRQn);    // TC2 ch0

    TC_Configure(TC2, 0,
        TC_CMR_WAVE | TC_CMR_WAVSEL_UP_RC | TC_CMR_TCCLKS_TIMER_CLOCK1);

    // CLOCK1 = MCK/2 = 84 MHz / 2 = 42 MHz
    uint32_t rc = 42000000UL / (uint32_t)sampleRate;
    TC_SetRC(TC2, 0, rc);

    TC2->TC_CHANNEL[0].TC_IER = TC_IER_CPCS;
    TC2->TC_CHANNEL[0].TC_IDR = ~TC_IER_CPCS;

    NVIC_EnableIRQ(TC6_IRQn);
    TC_Start(TC2, 0);
}

// ── Safe-clamp helpers ───────────────────────────────────────────────────────
float clampAmplitude(float a) {
    if (a < 0.0f)             a = 0.0f;
    if (a > MAX_AMPLITUDE)    a = MAX_AMPLITUDE;
    return a;
}

float clampDC(float dc) {
    if (dc < 0.0f)    dc = 0.0f;
    if (dc > MAX_DC)  dc = MAX_DC;
    return dc;
}

float clampFreq(float f) {
    if (f < 0.0f)       f = 0.0f;
    if (f > MAX_FREQ)   f = MAX_FREQ;
    return f;
}

float degreesToRad(float deg) {
    return deg * (float)PI / 180.0f;
}

// ── Serial command parser ────────────────────────────────────────────────────
char   g_cmdBuf[CMD_BUF_SIZE];
int    g_cmdIdx = 0;

void processCommand(const char* cmd) {
    char key[8];
    float val;

    if (strncmp(cmd, "START", 5) == 0) {
        g_sampleCount = 0;
        g_running = true;
        Serial.println("OK START");
        return;
    }

    if (strncmp(cmd, "STOP", 4) == 0) {
        g_running = false;
        analogWrite(DAC0, 0);
        Serial.println("OK STOP");
        return;
    }

    if (strncmp(cmd, "STATUS", 6) == 0) {
        // Print JSON — Python parses this
        Serial.print("{");
        Serial.print("\"a1\":"); Serial.print(g_amp1, 4); Serial.print(",");
        Serial.print("\"f1\":"); Serial.print(g_freq1, 4); Serial.print(",");
        Serial.print("\"p1\":"); Serial.print(g_phase1 * 180.0f / PI, 4); Serial.print(",");
        Serial.print("\"a2\":"); Serial.print(g_amp2, 4); Serial.print(",");
        Serial.print("\"f2\":"); Serial.print(g_freq2, 4); Serial.print(",");
        Serial.print("\"p2\":"); Serial.print(g_phase2 * 180.0f / PI, 4); Serial.print(",");
        Serial.print("\"dc\":"); Serial.print(g_dc, 4); Serial.print(",");
        Serial.print("\"sr\":"); Serial.print(g_sampleRate); Serial.print(",");
        Serial.print("\"running\":"); Serial.print(g_running ? "true" : "false");
        Serial.println("}");
        return;
    }

    // SET <KEY> <VALUE>
    if (sscanf(cmd, "SET %7s %f", key, &val) == 2) {

        if (strcmp(key, "A1") == 0) {
            g_amp1 = clampAmplitude(val);
            Serial.print("OK A1="); Serial.println(g_amp1, 4);

        } else if (strcmp(key, "A2") == 0) {
            g_amp2 = clampAmplitude(val);
            Serial.print("OK A2="); Serial.println(g_amp2, 4);

        } else if (strcmp(key, "F1") == 0) {
            g_freq1 = clampFreq(val);
            g_sampleCount = 0;
            Serial.print("OK F1="); Serial.println(g_freq1, 4);

        } else if (strcmp(key, "F2") == 0) {
            g_freq2 = clampFreq(val);
            g_sampleCount = 0;
            Serial.print("OK F2="); Serial.println(g_freq2, 4);

        } else if (strcmp(key, "P1") == 0) {
            g_phase1 = degreesToRad(val);
            Serial.print("OK P1="); Serial.println(val, 4);

        } else if (strcmp(key, "P2") == 0) {
            g_phase2 = degreesToRad(val);
            Serial.print("OK P2="); Serial.println(val, 4);

        } else if (strcmp(key, "DC") == 0) {
            g_dc = clampDC(val);
            Serial.print("OK DC="); Serial.println(g_dc, 4);

        } else if (strcmp(key, "SR") == 0) {
            bool was_running = g_running;
            g_running = false;
            delay(2);               // let ISR settle
            setupTimer((int)val);
            g_running = was_running;
            Serial.print("OK SR="); Serial.println(g_sampleRate);

        } else {
            Serial.print("ERR unknown key: "); Serial.println(key);
        }
        return;
    }

    Serial.print("ERR bad command: "); Serial.println(cmd);
}

// ── setup / loop ─────────────────────────────────────────────────────────────
void setup() {
    Serial.begin(115200);
    while (!Serial);            // wait for USB CDC

    analogWriteResolution(12);  // 12-bit DAC on Due
    analogWrite(DAC0, 0);

    setupTimer(g_sampleRate);

    Serial.println("READY dual_sine_dac v1.0");
}

void loop() {
    while (Serial.available()) {
        char c = (char)Serial.read();

        if (c == '\n' || c == '\r') {
            if (g_cmdIdx > 0) {
                g_cmdBuf[g_cmdIdx] = '\0';
                processCommand(g_cmdBuf);
                g_cmdIdx = 0;
            }
        } else {
            if (g_cmdIdx < CMD_BUF_SIZE - 1) {
                g_cmdBuf[g_cmdIdx++] = c;
            }
            // silently drop overflow characters
        }
    }
}
